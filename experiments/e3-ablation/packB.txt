# WIRING PACK v1 - generator-data form
# 'motif NAME(slots): template' defines a wiring shape; template lines use $slots.
# 'inst NAME(fills)' instantiates it: expand the template with the fills.
# A following indented '+ line' is a DELTA: after expansion, add that line verbatim.
# Expansion of every inst reproduces the explicit form exactly.

motif CommandHandler(H, S, f, m, entity, action):
  unit $H
  anno $H @CommandType entity=$entity action=$action
  dep $H.$f: $S
  edge $H -> $S#$m
